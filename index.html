function createAzeeziaChart() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); sheet.clear(); // Clear existing data // 1. Define Headers var headers = [ "Male Ward", "Female Ward", "OBG Ward", "Single Room", "Standard Room", "Deluxe Room", "Premium Room", "Suite Room" ]; // 2. Set Headers & Style var headerRange = sheet.getRange(1, 1, 1, 8); headerRange.setValues([headers]); headerRange.setBackground("#0E7490"); // Azeezia Teal headerRange.setFontColor("white"); headerRange.setFontWeight("bold"); headerRange.setHorizontalAlignment("center"); headerRange.setVerticalAlignment("middle"); // 3. Generate Data Rows (15 Patients) var data = []; for (var i = 0; i < 15; i++) { // Row 1: Details var row1 = []; for(var k=0; k<8; k++) row1.push("Pt No: ___ Room: ___"); data.push(row1); // Row 2: Name var row2 = []; for(var k=0; k<8; k++) row2.push("Name: ____________________"); data.push(row2); } // 4. Write Data to Sheet sheet.getRange(2, 1, data.length, 8).setValues(data); // 5. Apply Formatting to Data Area var fullRange = sheet.getRange(1, 1, data.length + 1, 8); fullRange.setBorder(true, true, true, true, true, true, "black", SpreadsheetApp.BorderStyle.SOLID); fullRange.setVerticalAlignment("middle"); // 6. Set Column Widths (approx 200 pixels) sheet.setColumnWidths(1, 8, 200); // 7. Resize Rows for better writing space sheet.setRowHeights(2, data.length, 30); }
